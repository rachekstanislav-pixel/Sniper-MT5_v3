void CheckAISignals() {
    if(!EnableAutoTrade) return;
    string file_name = "ai_signal.txt";
    int handle = FileOpen(file_name, FILE_READ|FILE_TXT|FILE_ANSI|FILE_COMMON);
    if(handle != INVALID_HANDLE) {
        string data = FileReadString(handle);
        FileClose(handle);
        FileDelete(file_name, FILE_COMMON); 
        string arr[];
        ushort sep = StringGetCharacter(";", 0);
        int elements = StringSplit(data, sep, arr);
        if(elements >= 3) {
            string sig_symbol = arr[0]; string action = arr[1]; string reason = arr[2];
            if(sig_symbol == _Symbol || sig_symbol == "ALL") {
                double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
                double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
                double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
                if(action == "BUY") {
                    double sl = NormalizeDouble(ask - StopLossPoints * point, _Digits);
                    double lot = CalculateSmartLot(ask, sl);
                    if(trade.Buy(lot, _Symbol, ask, sl, 0, "AI NEWS BUY")) {
                        DrawArrow("AI_BUY_" + IntegerToString(GetTickCount()), TimeCurrent(), ask - 20*point, clrAqua, 233, "AI BUY: " + reason, true);
                        SendTelegramPhotoAlert("🤖 AI СИГНАЛ НА ПОКУПКУ!\nПричина: " + reason);
                    }
                } 
                else if(action == "SELL") {
                    double sl = NormalizeDouble(bid + StopLossPoints * point, _Digits);
                    double lot = CalculateSmartLot(bid, sl);
                    if(trade.Sell(lot, _Symbol, bid, sl, 0, "AI NEWS SELL")) {
                        DrawArrow("AI_SELL_" + IntegerToString(GetTickCount()), TimeCurrent(), bid + 20*point, clrMagenta, 234, "AI SELL: " + reason, false);
                        SendTelegramPhotoAlert("🤖 AI СИГНАЛ НА ПРОДАЖУ!\nПричина: " + reason);
                    }
                }
            }
        }
    }
}

double GetDynamicRisk() {
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    if (balance <= 100.0) return 5.0;
    if (balance >= 1000.0) return 1.0;
    double risk = 5.0 - (balance - 100.0) * (4.0 / 900.0);
    return NormalizeDouble(risk, 2);
}

double GetDynamicGlobalRisk() {
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    if (balance <= 100.0) return 20.0;
    if (balance >= 1000.0) return 5.0;
    double g_risk = 20.0 - (balance - 100.0) * (15.0 / 900.0);
    return NormalizeDouble(g_risk, 2);
}

double CalculateSmartLot(double entry_price, double stop_loss) {
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double risk_amount = balance * (GetDynamicRisk() / 100.0);
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   double tick_size = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double tick_value = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   
   double sl_points = MathAbs(entry_price - stop_loss) / point;
   if(sl_points < 10) sl_points = 10; 
   
   double value_per_point = tick_value * (point / tick_size);
   double risk_per_lot = sl_points * value_per_point;
   double lot = risk_amount / risk_per_lot;
   
   double min_lot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double max_lot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double step = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   lot = MathRound(lot / step) * step;
   if(lot < min_lot) lot = min_lot;
   if(lot > max_lot) lot = max_lot;
   return lot;
}

bool CanOpenLot(int type, string block_comment) {
   if(!EnableAutoTrade) return false;
   double total_vol = 0;
   for(int i = 0; i < PositionsTotal(); i++) {
      ulong ticket = PositionGetTicket(i);
      if(PositionSelectByTicket(ticket) && PositionGetString(POSITION_SYMBOL) == _Symbol && PositionGetInteger(POSITION_MAGIC) == MagicNumber) {
         if(PositionGetString(POSITION_COMMENT) == block_comment) return false; 
         if(PositionGetInteger(POSITION_TYPE) == type) total_vol += PositionGetDouble(POSITION_VOLUME);
      }
   }
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double max_vol = MathMax(SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN), MathRound((balance / 10000.0) / SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP)) * SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP));
   return (total_vol < max_vol);
}

void ManagePositions() {
   double buy_vol = 0, sell_vol = 0, buy_weighted = 0, sell_weighted = 0;
   bool has_buy = false, has_sell = false;

   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double tick_val = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double tick_size = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);

   for(int i = PositionsTotal() - 1; i >= 0; i--) {
      ulong ticket = PositionGetTicket(i);
      if(PositionSelectByTicket(ticket) && PositionGetString(POSITION_SYMBOL) == _Symbol && (PositionGetInteger(POSITION_MAGIC) == 0 || PositionGetInteger(POSITION_MAGIC) == MagicNumber)) {
         double vol = PositionGetDouble(POSITION_VOLUME);
         double open_price = PositionGetDouble(POSITION_PRICE_OPEN);
         long type = PositionGetInteger(POSITION_TYPE);
         if(type == POSITION_TYPE_BUY) { buy_vol += vol; buy_weighted += (open_price * vol); has_buy = true; } 
         else if(type == POSITION_TYPE_SELL) { sell_vol += vol; sell_weighted += (open_price * vol); has_sell = true; }
      }
   }
   
   if(has_buy && buy_vol > 0) {
      double avg_buy = buy_weighted / buy_vol;
      double profit_pts = (bid - avg_buy) / point;
      if(profit_pts >= Physical_BE_Trigger) {
         double new_sl = NormalizeDouble(avg_buy + Physical_BE_Level * point, _Digits);
         for(int i = PositionsTotal() - 1; i >= 0; i--) {
            ulong ticket = PositionGetTicket(i);
            if(PositionSelectByTicket(ticket) && PositionGetString(POSITION_SYMBOL) == _Symbol && (PositionGetInteger(POSITION_MAGIC) == 0 || PositionGetInteger(POSITION_MAGIC) == MagicNumber) && PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY) {
               double current_sl = PositionGetDouble(POSITION_SL);
               double current_tp = PositionGetDouble(POSITION_TP);
               if(current_sl < new_sl || current_sl == 0) trade.PositionModify(ticket, new_sl, current_tp);
            }
         }
      }
      double target_tp_buy = NormalizeDouble(avg_buy + Fixed_TakeProfit * point, _Digits);
      string line_buy = "Virtual_TP_BUY";
      if(ObjectFind(0, line_buy) < 0) ObjectCreate(0, line_buy, OBJ_HLINE, 0, 0, target_tp_buy);
      ObjectSetDouble(0, line_buy, OBJPROP_PRICE, 0, target_tp_buy);
      ObjectSetInteger(0, line_buy, OBJPROP_COLOR, clrLime);
      ObjectSetString(0, line_buy, OBJPROP_TEXT, " Basket TP BUY");
      
      double val_per_point_buy = buy_vol * tick_val * (point / tick_size);
      if(val_per_point_buy > 0) {
         double pts_to_stop_buy = (balance * (GetDynamicGlobalRisk() / 100.0)) / val_per_point_buy;
         double global_stop_buy = NormalizeDouble(avg_buy - pts_to_stop_buy * point, _Digits);
         string gstop_buy = "Global_Stop_BUY";
         if(ObjectFind(0, gstop_buy) < 0) ObjectCreate(0, gstop_buy, OBJ_HLINE, 0, 0, global_stop_buy);
         ObjectSetDouble(0, gstop_buy, OBJPROP_PRICE, 0, global_stop_buy);
         ObjectSetInteger(0, gstop_buy, OBJPROP_COLOR, clrFireBrick);
         ObjectSetString(0, gstop_buy, OBJPROP_TEXT, " Global Stop BUY");
      }
      if(bid >= target_tp_buy) {
         for(int i = PositionsTotal() - 1; i >= 0; i--) {
            ulong ticket = PositionGetTicket(i);
            if(PositionSelectByTicket(ticket) && PositionGetString(POSITION_SYMBOL) == _Symbol && (PositionGetInteger(POSITION_MAGIC) == 0 || PositionGetInteger(POSITION_MAGIC) == MagicNumber) && PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY) trade.PositionClose(ticket);
         }
         ObjectDelete(0, "Virtual_TP_BUY"); ObjectDelete(0, "Global_Stop_BUY");
      }
   } else { ObjectDelete(0, "Virtual_TP_BUY"); ObjectDelete(0, "Global_Stop_BUY"); }
   
   if(has_sell && sell_vol > 0) {
      double avg_sell = sell_weighted / sell_vol;
      double profit_pts = (avg_sell - ask) / point;
      if(profit_pts >= Physical_BE_Trigger) {
         double new_sl = NormalizeDouble(avg_sell - Physical_BE_Level * point, _Digits);
         for(int i = PositionsTotal() - 1; i >= 0; i--) {
            ulong ticket = PositionGetTicket(i);
            if(PositionSelectByTicket(ticket) && PositionGetString(POSITION_SYMBOL) == _Symbol && (PositionGetInteger(POSITION_MAGIC) == 0 || PositionGetInteger(POSITION_MAGIC) == MagicNumber) && PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_SELL) {
               double current_sl = PositionGetDouble(POSITION_SL);
               double current_tp = PositionGetDouble(POSITION_TP);
               if(current_sl > new_sl || current_sl == 0) trade.PositionModify(ticket, new_sl, current_tp);
            }
         }
      }
      double target_tp_sell = NormalizeDouble(avg_sell - Fixed_TakeProfit * point, _Digits);
      string line_sell = "Virtual_TP_SELL";
      if(ObjectFind(0, line_sell) < 0) ObjectCreate(0, line_sell, OBJ_HLINE, 0, 0, target_tp_sell);
      ObjectSetDouble(0, line_sell, OBJPROP_PRICE, 0, target_tp_sell);
      ObjectSetInteger(0, line_sell, OBJPROP_COLOR, clrRed);
      ObjectSetString(0, line_sell, OBJPROP_TEXT, " Basket TP SELL");
      
      double val_per_point_sell = sell_vol * tick_val * (point / tick_size);
      if(val_per_point_sell > 0) {
         double pts_to_stop_sell = (balance * (GetDynamicGlobalRisk() / 100.0)) / val_per_point_sell;
         double global_stop_sell = NormalizeDouble(avg_sell + pts_to_stop_sell * point, _Digits);
         string gstop_sell = "Global_Stop_SELL";
         if(ObjectFind(0, gstop_sell) < 0) ObjectCreate(0, gstop_sell, OBJ_HLINE, 0, 0, global_stop_sell);
         ObjectSetDouble(0, gstop_sell, OBJPROP_PRICE, 0, global_stop_sell);
         ObjectSetInteger(0, gstop_sell, OBJPROP_COLOR, clrFireBrick);
         ObjectSetString(0, gstop_sell, OBJPROP_TEXT, " Global Stop SELL");
      }
      if(ask <= target_tp_sell) {
         for(int i = PositionsTotal() - 1; i >= 0; i--) {
            ulong ticket = PositionGetTicket(i);
            if(PositionSelectByTicket(ticket) && PositionGetString(POSITION_SYMBOL) == _Symbol && (PositionGetInteger(POSITION_MAGIC) == 0 || PositionGetInteger(POSITION_MAGIC) == MagicNumber) && PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_SELL) trade.PositionClose(ticket);
         }
         ObjectDelete(0, "Virtual_TP_SELL"); ObjectDelete(0, "Global_Stop_SELL");
      }
   } else { ObjectDelete(0, "Virtual_TP_SELL"); ObjectDelete(0, "Global_Stop_SELL"); }
}

void CheckEquityProtection() {
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double equity = AccountInfoDouble(ACCOUNT_EQUITY);
   if(balance > 0 && ((balance - equity) / balance * 100.0) >= GetDynamicGlobalRisk()) {
      for(int i = PositionsTotal() - 1; i >= 0; i--) {
         ulong ticket = PositionGetTicket(i);
         if(PositionSelectByTicket(ticket) && PositionGetString(POSITION_SYMBOL) == _Symbol) trade.PositionClose(ticket);
      }
      SendTelegramPhotoAlert("⚠️ ГЛОБАЛЬНЫЙ СТОП СРАБОТАЛ! Закрыта пачка сделок.");
   }
}

double GetVolumeSMA(int period) {
    long vol[];
    if(CopyTickVolume(_Symbol, _Period, 1, period, vol) < period) return 0.0;
    double sum = 0.0;
    for(int i=0; i<period; i++) sum += (double)vol[i];
    return sum / (double)period;
}

   if(CopyBuffer(h_atr, 0, 0, 1, atr_val) <= 0) return;
   if(CopyBuffer(h_atr_ma, 0, 0, 1, atr_ma_val) <= 0) return;

   if(atr_val[0] < atr_ma_val[0]) {

      double min_l = iLow(_Symbol, _Period, iLowest(_Symbol, _Period, MODE_LOW, 3, 0));

      string direction = G_Fibo_isUp ? "BUY" : "SELL";

      double txt_price = G_Fibo_isUp ? min_l : max_h;

      string comp_name = "CompZone_Current";

      if(!G_Compression_Active) {
         G_Compression_Active = true;
         
         // МГНОВЕННЫЙ ВХОД НА ОТРИСОВКЕ ЗОНЫ

            if(CanOpenLot(POSITION_TYPE_BUY, "SMC_Comp") && TimeCurrent() - G_LastBuyTime >= EntryDelayMinutes * 60) {

               double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);

// МОДУЛЬ 4: ТОРГОВЛЯ ОТ EMA (LIVE)
void CheckEMATouchLive() {
   if(!EnableAutoTrade) return;
   double current_ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double current_bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   
   double e21[], e55[], e200[];
   if(CopyBuffer(h_ema21, 0, 0, 1, e21) <= 0 || CopyBuffer(h_ema55, 0, 0, 1, e55) <= 0 || CopyBuffer(h_ema200, 0, 0, 1, e200) <= 0) return;
   
   bool allow_200 = true;
   bool allow_55 = (_Period >= PERIOD_M15);
   bool allow_21 = (_Period >= PERIOD_H1);
   
   bool uptrend = (e21[0] > e55[0] && e55[0] > e200[0]);
   bool downtrend = (e21[0] < e55[0] && e55[0] < e200[0]);
   double tolerance = 2 * _Point;
   
   if(uptrend && TimeCurrent() - G_LastBuyTime >= EntryDelayMinutes*60) {
      bool touched = false; double sl = 0;
      if(allow_21 && MathAbs(current_ask - e21[0]) <= tolerance) { touched = true; sl = NormalizeDouble(e55[0], _Digits); }
      else if(allow_55 && MathAbs(current_ask - e55[0]) <= tolerance) { touched = true; sl = NormalizeDouble(e200[0], _Digits); }
      else if(allow_200 && MathAbs(current_ask - e200[0]) <= tolerance) { touched = true; sl = NormalizeDouble(e200[0] - 150*_Point, _Digits); }
      
      if(touched && CanOpenLot(POSITION_TYPE_BUY, "SMC_EMA")) {
         double lot = CalculateSmartLot(current_ask, sl);
         if(trade.Buy(lot, _Symbol, current_ask, sl, 0, "SMC_EMA")) G_LastBuyTime = TimeCurrent();
      }
   }
   
   if(downtrend && TimeCurrent() - G_LastSellTime >= EntryDelayMinutes*60) {
      bool touched = false; double sl = 0;
      if(allow_21 && MathAbs(current_bid - e21[0]) <= tolerance) { touched = true; sl = NormalizeDouble(e55[0], _Digits); }
      else if(allow_55 && MathAbs(current_bid - e55[0]) <= tolerance) { touched = true; sl = NormalizeDouble(e200[0], _Digits); }
      else if(allow_200 && MathAbs(current_bid - e200[0]) <= tolerance) { touched = true; sl = NormalizeDouble(e200[0] + 150*_Point, _Digits); }
      
      if(touched && CanOpenLot(POSITION_TYPE_SELL, "SMC_EMA")) {
         double lot = CalculateSmartLot(current_bid, sl);
         if(trade.Sell(lot, _Symbol, current_bid, sl, 0, "SMC_EMA")) G_LastSellTime = TimeCurrent();
      }
   }
}

// МОДУЛЬ 2: ПРОБОЙ БАЗЫ/НАКОПЛЕНИЯ
void CheckBaseBreakout() {
   double h8 = iHigh(_Symbol, _Period, iHighest(_Symbol, _Period, MODE_HIGH, 8, 1));
   double l8 = iLow(_Symbol, _Period, iLowest(_Symbol, _Period, MODE_LOW, 8, 1));
   double range = h8 - l8;
   
   double atr[];
   if(CopyBuffer(h_atr, 0, 1, 1, atr) <= 0) return;
   
   // Проверка узости базы (эффект сжатой пружины)
   if (range > 0 && range < atr[0] * 1.5) {
      string b_name = "BaseZone_" + TimeToString(iTime(_Symbol, _Period, 8));
      DrawR(b_name, iTime(_Symbol, _Period, 8), h8, TimeCurrent() + PeriodSeconds()*10, l8, clrDarkGray, "BASE");
      
      // Детект импульсного пробоя + объема
      double c1 = iClose(_Symbol, _Period, 1);
      long vol1 = iVolume(_Symbol, _Period, 1);
      double sma_v = GetVolumeSMA(Vol_SMA_Period);
      double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      
      if (c1 > h8 && (double)vol1 > sma_v * VolSpikeCoef) { // Bullish Breakout
         if(EnableAutoTrade && CanOpenLot(POSITION_TYPE_BUY, "SMC_Base") && TimeCurrent() - G_LastBuyTime >= EntryDelayMinutes*60) {
             double sl = NormalizeDouble(l8 - 15*_Point, _Digits);
             double lot = CalculateSmartLot(ask, sl);
             if(trade.Buy(lot, _Symbol, ask, sl, 0, "SMC_Base")) {
                 G_LastBuyTime = TimeCurrent();
             }
         }
      }
      if (c1 < l8 && (double)vol1 > sma_v * VolSpikeCoef) { // Bearish Breakout
         if(EnableAutoTrade && CanOpenLot(POSITION_TYPE_SELL, "SMC_Base") && TimeCurrent() - G_LastSellTime >= EntryDelayMinutes*60) {
             double sl = NormalizeDouble(h8 + 15*_Point, _Digits);
             double lot = CalculateSmartLot(bid, sl);
             if(trade.Sell(lot, _Symbol, bid, sl, 0, "SMC_Base")) {
                 G_LastSellTime = TimeCurrent();
             }
         }
      }
   }
}

void UpdateMTFSwings() {
   for(int i = 0; i < 6; i++) {
      int limit = 150; 
      int last_sh = -1, last_sl = -1;
      
      for(int j = Fractal_Strength + 1; j < limit; j++) {
         bool isHigh = true, isLow = true;
         for(int k = 1; k <= Fractal_Strength; k++) {
            if(iHigh(_Symbol, mtf_tfs[i], j - k) >= iHigh(_Symbol, mtf_tfs[i], j) || iHigh(_Symbol, mtf_tfs[i], j + k) >= iHigh(_Symbol, mtf_tfs[i], j)) isHigh = false;
            if(iLow(_Symbol, mtf_tfs[i], j - k) <= iLow(_Symbol, mtf_tfs[i], j) || iLow(_Symbol, mtf_tfs[i], j + k) <= iLow(_Symbol, mtf_tfs[i], j)) isLow = false;
         }
         if(last_sh == -1 && isHigh) last_sh = j;
         if(last_sl == -1 && isLow) last_sl = j;
         if(last_sh != -1 && last_sl != -1) break;
      }
      if(last_sh != -1) { G_MTF_SwingHigh[i] = iHigh(_Symbol, mtf_tfs[i], last_sh); G_MTF_SwingHighTime[i] = iTime(_Symbol, mtf_tfs[i], last_sh); } 
      else { G_MTF_SwingHigh[i] = 0; G_MTF_SwingHighTime[i] = 0; }
      
      if(last_sl != -1) { G_MTF_SwingLow[i] = iLow(_Symbol, mtf_tfs[i], last_sl); G_MTF_SwingLowTime[i] = iTime(_Symbol, mtf_tfs[i], last_sl); } 
      else { G_MTF_SwingLow[i] = 0; G_MTF_SwingLowTime[i] = 0; }
   }
}

string GetNearestOBDistance() {
    double current_price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
    double min_dist = 999999;
    int total = ObjectsTotal(0, 0, OBJ_RECTANGLE);
    for(int i = 0; i < total; i++) {
        string name = ObjectName(0, i, 0, OBJ_RECTANGLE);
        if(StringFind(name, "OB_") == 0) {
            double p1 = ObjectGetDouble(0, name, OBJPROP_PRICE, 0);
            double p2 = ObjectGetDouble(0, name, OBJPROP_PRICE, 1);
            double top = MathMax(p1, p2), bot = MathMin(p1, p2);
            double dist = 0;
            if(current_price > top) dist = current_price - top;
            else if(current_price < bot) dist = bot - current_price;
            if(dist < min_dist) min_dist = dist;
        }
    }
    if(min_dist == 999999) return "NONE";
    return IntegerToString((int)(min_dist / _Point)) + " pts";
}

double GetDailyPNL() {
   double daily_profit = 0;
   HistorySelect(iTime(_Symbol, PERIOD_D1, 0), TimeCurrent());
   for(int i = 0; i < HistoryDealsTotal(); i++) {

         daily_profit += HistoryDealGetDouble(ticket, DEAL_SWAP);
         daily_profit += HistoryDealGetDouble(ticket, DEAL_COMMISSION);
      }
   }
   return daily_profit;
}

void DrawStructureAndFibo() {
   ENUM_TIMEFRAMES tf = _Period;
   if(tf == PERIOD_M1 || tf == PERIOD_M5) tf = PERIOD_M15;
   if(G_Struct_Valid) {
       double c1 = iClose(_Symbol, tf, 1);
       if(c1 < G_Struct_High && c1 > G_Struct_Low) return; 
   }

   int limit = 60;
   switch(tf) { case PERIOD_M15: limit = 200; break; case PERIOD_H1: limit = 100; break; case PERIOD_H4: limit = 100; break; case PERIOD_D1: limit = 100; break; default: limit = 200; break; }
   
   int last_sh = -1, last_sl = -1;
   for(int i = Fractal_Strength + 1; i < limit; i++) {
      if(last_sh == -1 && IsSwingHigh(tf, i, Fractal_Strength)) last_sh = i;
      if(last_sl == -1 && IsSwingLow(tf, i, Fractal_Strength)) last_sl = i;
      if(last_sh != -1 && last_sl != -1) break;
   }
   if(last_sh == -1 || last_sl == -1) { G_Struct_Valid = false; return; }
   
   double sh_price = iHigh(_Symbol, tf, last_sh); double sl_price = iLow(_Symbol, tf, last_sl);
   bool isUp = (last_sl > last_sh);
   double last_closed_price = iClose(_Symbol, tf, 1);
   string struct_text = ""; color struct_color = clrNONE;
   
   if(isUp && last_closed_price > sh_price) { 
      struct_text = (iLow(_Symbol, tf, last_sh+10) > sl_price) ? "BOS" : "CHOCH"; 
      struct_color = (struct_text == "BOS") ? Color_BOS : Color_CHOCH;
      CreateStructLine("StructLine", iTime(_Symbol, tf, last_sh), sh_price, iTime(_Symbol, tf, 1), sh_price, struct_color, struct_text);
   } 
   else if(!isUp && last_closed_price < sl_price) { 
      struct_text = (iHigh(_Symbol, tf, last_sl+10) < sh_price) ? "BOS" : "CHOCH";
      struct_color = (struct_text == "BOS") ? Color_BOS : Color_CHOCH;
      CreateStructLine("StructLine", iTime(_Symbol, tf, last_sl), sl_price, iTime(_Symbol, tf, 1), sl_price, struct_color, struct_text);
   }

   double r = sh_price - sl_price;
   G_Struct_High = sh_price; G_Struct_Low = sl_price;
   G_Fibo_EQ = isUp ? sh_price - r * 0.5 : sl_price + r * 0.5;
   G_Fibo_618 = isUp ? sh_price - r * 0.618 : sl_price + r * 0.618;
   G_Fibo_790 = isUp ? sh_price - r * 0.79 : sl_price + r * 0.79;
   G_Fibo_isUp = isUp; G_Struct_Valid = true;
   
   CreateFiboLine("Line_000", isUp ? sh_price : sl_price, clrGray, STYLE_SOLID, "0.0");
   CreateFiboLine("Line_050", G_Fibo_EQ, clrGray, STYLE_SOLID, "0.5 EQ");
   CreateFiboLine("Line_618", G_Fibo_618, clrGold, STYLE_SOLID, "0.618 Golden");
   CreateFiboLine("Line_Dia2", G_Fibo_790, Diamond_Color, STYLE_SOLID, "0.790 OTE");
   CreateFiboLine("Line_100", isUp ? sl_price : sh_price, clrGray, STYLE_SOLID, "1.0");
}

bool IsSwingHigh(ENUM_TIMEFRAMES tf, int index, int strength) { 
   for(int i = 1; i <= strength; i++) {
      if(iHigh(_Symbol, tf, index - i) >= iHigh(_Symbol, tf, index) || iHigh(_Symbol, tf, index + i) >= iHigh(_Symbol, tf, index)) return false;
   }
   return true; 
}
bool IsSwingLow(ENUM_TIMEFRAMES tf, int index, int strength) { 
   for(int i = 1; i <= strength; i++) {
      if(iLow(_Symbol, tf, index - i) <= iLow(_Symbol, tf, index) || iLow(_Symbol, tf, index + i) <= iLow(_Symbol, tf, index)) return false;
   }
   return true; 
}

string GetActiveMacroOTE(double cur) {
   for(int i = 5; i >= 3; i--) {
      if(G_MTF_SwingHigh[i] <= 0 || G_MTF_SwingLow[i] <= 0) continue;
      double r = G_MTF_SwingHigh[i] - G_MTF_SwingLow[i];
      if(r <= 0) continue;
      
      if(G_MTF_SwingLow[i] < G_MTF_SwingHigh[i]) {
         double ote_top = G_MTF_SwingHigh[i] - r * 0.618;
         double ote_bot = G_MTF_SwingHigh[i] - r * 0.790;
         if(cur >= ote_bot && cur <= ote_top) return mtf_names[i] + " BUY";
      } else {
         double ote_bot = G_MTF_SwingHigh[i] + r * 0.618;
         double ote_top = G_MTF_SwingHigh[i] + r * 0.790;
         if(cur >= ote_bot && cur <= ote_top) return mtf_names[i] + " SELL";
      }
   }
   return "NONE";
}

   double c1 = iClose(_Symbol, _Period, 1);
   int total = ObjectsTotal(0, 0, OBJ_RECTANGLE);
   for(int i = total - 1; i >= 0; i--) {
      string name = ObjectName(0, i, 0, OBJ_RECTANGLE);
      if(StringFind(name, "OTE_Buy") == 0 || StringFind(name, "OB_H4_B") == 0 || StringFind(name, "FVG_H4_B") == 0) {
         double bot = MathMin(ObjectGetDouble(0, name, OBJPROP_PRICE, 0), ObjectGetDouble(0, name, OBJPROP_PRICE, 1));
         if(c1 < bot) { ObjectDelete(0, name); ObjectDelete(0, "Txt_" + name); }
      }
      else if(StringFind(name, "OTE_Sell") == 0 || StringFind(name, "OB_H4_S") == 0 || StringFind(name, "FVG_H4_S") == 0) {
         double top = MathMax(ObjectGetDouble(0, name, OBJPROP_PRICE, 0), ObjectGetDouble(0, name, OBJPROP_PRICE, 1));
         if(c1 > top) { ObjectDelete(0, name); ObjectDelete(0, "Txt_" + name); }
      }
   }
}
